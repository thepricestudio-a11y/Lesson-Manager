async function initApp() { try { const [pupilsRes, locsRes, rulesRes] = await Promise.all([ window.supabaseClient.from('pupils').select('*').order('pupil_name'), window.supabaseClient.from('locations').select('*').order('location_name'), window.supabaseClient.from('attendance_rules').select('*').order('id') ]); // NEW: Force database errors to pop up on the screen if (pupilsRes.error) alert("Database Error (Pupils): " + pupilsRes.error.message); if (locsRes.error) alert("Database Error (Locations): " + locsRes.error.message); pupils = pupilsRes.data || []; locations = locsRes.data || []; attendanceRules = rulesRes.data || []; const pupilSelect = document.getElementById('pupilSelect'); pupils.forEach(p => { let opt = document.createElement('option'); opt.value = p.pupil_id; opt.textContent = p.pupil_name; pupilSelect.appendChild(opt); }); const locSelect = document.getElementById('locationSelect'); locations.forEach(l => { let opt = document.createElement('option'); opt.value = l.id; opt.textContent = l.location_name; locSelect.appendChild(opt); }); const attSelect = document.getElementById('attendanceSelect'); attendanceRules.forEach(a => { let opt = document.createElement('option'); opt.value = a.id; opt.textContent = a.description; attSelect.appendChild(opt); if(a.description.toLowerCase().includes('present')) { opt.selected = true; } }); } catch (error) { alert("Critical Initialization Error: " + error.message); console.error("Error during initialization:", error); } }
